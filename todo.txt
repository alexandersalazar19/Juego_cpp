        const char* t=TextFormat("%d",(int)ceil(restante));
        DrawText(t,(int)pos.x-MeasureText(t,20)/2,(int)pos.y-10,20,WHITE);
    }
}

BombChallenge::BombChallenge(const ConfigDificultad& cfg):cada(cfg.bombaCada),proxima(variar(cfg.bombaCada)){}

void BombChallenge::update(float dt,bool activo,Context& ctx){
    if(activo && cada>0){
        proxima-=dt;
        if(proxima<=0){
            bombas.push_back({Arena::puntoLibre(ctx.jugador.getPos(),200.0F,40.0F),DURACION});
            proxima=variar(cada);
        }
    }

    for(auto it=bombas.begin();it!=bombas.end();){
        it->restante-=dt;
        if(CheckCollisionCircles(it->pos,RADIO,ctx.jugador.getPos(),ctx.jugador.getRadio())){
            ctx.puntos+=puntosPor(150,ctx.cfg,ctx.nivel);
            it=bombas.erase(it);
        }else if(it->restante<=0){
            ctx.jugador.recibirGolpe();
            it=bombas.erase(it);
        }else it++;
    }
}

void BombChallenge::dibujarMundo() const {
    for(const auto& b:bombas) dibujarBomba(b.pos,b.restante);
}

void BombChallenge::dibujarCapa(const Context&) const {
    if(bombas.empty()) return;
    float urgencia=0.0F; //la de la bomba mas cercana a explotar
    for(const auto& b:bombas) urgencia=max(urgencia,1.0F-b.restante/DURACION);
    dibujarAlertaBomba(urgencia);
}

void Player::update(float dt){
    Vector2 dir={0.0F,0.0F};
    if(IsKeyDown(KEY_D)) dir.x++;
    if(IsKeyDown(KEY_A)) dir.x--;
    if(IsKeyDown(KEY_S)) dir.y++;
    if(IsKeyDown(KEY_W)) dir.y--;

    //Hacer que no se mueva mas rapido en diagonal
    float largo=sqrt(dir.x*dir.x + dir.y*dir.y);
    if(largo>0){
        dir.x/=largo;
        dir.y/=largo;
    }

    pos.x+=dir.x*velocidad*dt;
    pos.y+=dir.y*velocidad*dt;

    //Limites de la ventana
    Rectangle a=Arena::area();
    pos.x=clamp(pos.x,a.x+radio,a.x+a.width-radio);
    pos.y=clamp(pos.y,a.y+radio,a.y+a.height-radio);

    if(invul>0) invul-=dt;
    if(cargas<MAX_CARGAS){
        recarga+=dt;
        if(recarga>=TIEMPO_RECARGA){
            cargas++;
            recarga=0.0F;
        }
    }
    if(cargas>0 && IsMouseButtonPressed(MOUSE_BUTTON_LEFT) && CheckCollisionPointRec(GetMousePosition(),Arena::area())){
        pos=destinoTp();
        cargas--;
        invul=max(invul,INVUL_TP);
    }
}

void Player::draw() const {
    Color c=MAROON;
    if(invul>0 && !((int)(invul*10)%2)) c=Fade(MAROON,0.35F); //parpadeo
    if(cargas>0 && CheckCollisionPointRec(GetMousePosition(),Arena::area())){
        if(ALCANCE_TP>0) DrawCircleLinesV(pos,ALCANCE_TP,Fade(GRAY,0.5F));
        DrawCircleLinesV(destinoTp(),radio,Fade(MAROON,0.6F));
    }
    DrawCircleV(pos,radio,c);
}

Vector2 Player::destinoTp() const {
    Rectangle a=Arena::area();
    Vector2 m=GetMousePosition();
    if(ALCANCE_TP>0){
        float dx=m.x-pos.x;
        float dy=m.y-pos.y;
        float d=std::sqrt(dx*dx+dy*dy);
        if(d>ALCANCE_TP){
            m.x=pos.x+dx/d*ALCANCE_TP;
            m.y=pos.y+dy/d*ALCANCE_TP;
        }
    }
    m.x=std::clamp(m.x,a.x+radio,a.x+a.width-radio);
    m.y=std::clamp(m.y,a.y+radio,a.y+a.height-radio);
    return m;
}

#include "game/Levels.h"
#include "game/Difficulty.h"
#include "raylib.h"
using namespace std;

namespace{
    const float DURACION_ANUNCIO=2.5F;
    const float DURACION_NIVEL=30.0F;
    const float RAMPA_MIN=0.7F; //velocidad al empezar el nivel
    const float RAMPA_MAX=1.3F; //velocidad al terminarlo
    const int MAX_COMBO=2; //retos combinados en Extremo

    const int GUION[]={MOD_BOMBA,MOD_TENUE,MOD_OSCURO,MOD_CALOR};
    const char* NOMBRES[]={"Bombas","Poca Visibilidad","Oscuridad","Sobrecalentamiento"};
    const int N_MODS=sizeof(GUION)/sizeof(GUION[0]);

    const int PROHIBIDOS[]={MOD_TENUE|MOD_OSCURO};

    bool prohibido(int mods){
        for(int p:PROHIBIDOS) if((mods&p)==p) return true;
        return false;
    }

    int modsParaNivel(int nivel,int dificultad,int previo){
        if(nivel<=0) return MOD_NINGUNO;
        if(nivel<=N_MODS) return GUION[nivel-1];
        if(dificultad<NUM_DIFICULTADES-1){ //de facil a dificil: uno solo
            int m;
            do{
                m=GUION[GetRandomValue(0,N_MODS-1)];
            }while(m==previo && N_MODS>1);
            return m;
        }
        //Extremo: combinacion aleatoria
        int k=GetRandomValue(1,MAX_COMBO);
        int mods=MOD_NINGUNO;
        int intentos=0;
        while(k>0 && intentos<30){
            intentos++;
            int m=GUION[GetRandomValue(0,N_MODS-1)];
            if(mods&m) continue; //ya lo tenia
            if(prohibido(mods|m)) continue;
            mods|=m;
            k--;
        }
        return mods;
    }
}

ControlNiveles::ControlNiveles():nivel(0),dificultad(0),mods(MOD_NINGUNO),fase(FaseNivel::Anuncio),tiempoFase(0.0F){}

void ControlNiveles::reiniciar(int d){
    nivel=0;
    dificultad=d;
    mods=modsParaNivel(0,d,MOD_NINGUNO);
    fase=FaseNivel::Anuncio;
    tiempoFase=0.0F;
}

void ControlNiveles::siguienteNivel(){
    int previo=mods;
    nivel++;
    mods=modsParaNivel(nivel,dificultad,previo);
    fase=FaseNivel::Anuncio;
    tiempoFase=0.0F;
}

bool ControlNiveles::update(float dt,bool hayObstaculos){
    tiempoFase+=dt;
    switch(fase){
        case FaseNivel::Anuncio:
            if(tiempoFase>=DURACION_ANUNCIO){
                fase=FaseNivel::Activo;
                tiempoFase=0.0F;
            }
            break;
        case FaseNivel::Activo:
            if(tiempoFase>=DURACION_NIVEL){
                fase=FaseNivel::Cierre;
                tiempoFase=0.0F;
            }
            break;
        case FaseNivel::Cierre:
            if(!hayObstaculos){
                siguienteNivel();
                return true;
            }
            break;
    }
    return false;
}

bool ControlNiveles::puedeSpawnear() const{
    return fase==FaseNivel::Activo;
}

string nombreMods(int mods){

    string s;
    for(int i=0;i<N_MODS;i++){
        if(mods&GUION[i]){
            if(!s.empty()) s+=" + ";
            s+=NOMBRES[i];
        }
    }
    return s;
}

    if(fase==FaseNivel::Anuncio) return RAMPA_MIN;

    void paredSimple(ListaObstaculos& out,int lado,float vel){
        const Rectangle a=Arena::area();
        const float grosor=aleatorio(GROSOR_MIN,GROSOR_MAX);
        const float largo=aleatorio(LARGO_MIN,LARGO_MAX);


        float x=(float)GetRandomValue((int)a.x,(int)(a.x+a.width-largo));
        float y=(float)GetRandomValue((int)a.y,(int)(a.y+a.height-largo));
        float der=a.x+a.width;
        float abajo=a.y+a.height;

        switch(lado){
            case 0: out.push_back(make_unique<Wall>(Rectangle{a.x-grosor,y,grosor,largo},Vector2{vel,0})); break;

        float inicioHueco=(float)GetRandomValue(40,(int)(largoLado-hueco-40));
        float finHueco=inicioHueco+hueco;

        Rectangle r1,r2;
        Vector2 v;
        if(vertical){
            float x=(!lado) ? a.x-grosor : a.x+a.width;
            v={(!lado) ? vel : -vel,0.0F};
            r1={x,a.y,grosor,inicioHueco};
            r2={x,a.y+finHueco,grosor,largoLado-finHueco};
        }else{
            float y=(lado==2) ? a.y-grosor : a.y+a.height;
            v={0.0F,(lado==2) ? vel : -vel};
            r1={a.x,y,inicioHueco,grosor};
            r2={a.x+finHueco,y,largoLado-finHueco,grosor};
        }

        out.push_back(make_unique<Wall>(r1,v));
        out.push_back(make_unique<Wall>(r2,v));
    }

    void paredDiagonal(ListaObstaculos& out,int lado,float vel){
        const Rectangle a=Arena::area();
        const Vector2 tam={aleatorio(LARGO_MIN,LARGO_MAX),aleatorio(GROSOR_MIN,GROSOR_MAX)};
        const float R=0.5F*sqrt(tam.x*tam.x + tam.y*tam.y);

        float x=(float)GetRandomValue((int)a.x,(int)(a.x+a.width));
        float y=(float)GetRandomValue((int)a.y,(int)(a.y+a.height));
        float midX=a.x+a.width/2;
        float midY=a.y+a.height/2;

        //apunta al centro de la pantalla para que no se salga de inmediato
        Vector2 pos,dir;
        switch(lado){
            case 0: pos={a.x-R,y}; dir={1,(y<midY) ? 1.0F : -1.0F}; break;
            case 1: pos={a.x+a.width+R,y}; dir={-1,(y<midY) ? 1.0F : -1.0F}; break;
            case 2: pos={x,a.y-R}; dir={(x<midX) ? 1.0F : -1.0F,1}; break;
            default: pos={x,a.y+a.height+R}; dir={(x<midX) ? 1.0F : -1.0F,-1}; break;
        }

        Vector2 v={dir.x*vel*0.7071F,dir.y*vel*0.7071F};

        float anguloMov=atan2(v.y,v.x)*RAD2DEG;
        float angulo=anguloMov+90.0F+(float)GetRandomValue(-30,30);

        out.push_back(make_unique<DiagonalWall>(pos,tam,angulo,v));
    }

    void circuloOnda(ListaObstaculos& out,int lado,float vel){
        const Rectangle a=Arena::area();
        const float r=aleatorio(RADIO_MIN,RADIO_MAX);
        float x=(float)GetRandomValue((int)a.x,(int)(a.x+a.width));
        float y=(float)GetRandomValue((int)a.y,(int)(a.y+a.height));

        float amplitud=(float)GetRandomValue(40,120);
        float frecuencia=GetRandomValue(20,40)/10.0F; //entre 2.0 y 4.0 rad/s

        Vector2 pos,v;
        switch(lado){
            case 0: pos={a.x-r,y}; v={vel,0}; break;
            case 1: pos={a.x+a.width+r,y}; v={-vel,0}; break;
            case 2: pos={x,a.y-r}; v={0,vel}; break;
            default: pos={x,a.y+a.height+r}; v={0,-vel}; break;
        }
        out.push_back(make_unique<CircleObstacle>(pos,r,v,amplitud,frecuencia));
    }

    int elegirTipo(const ConfigDificultad& c){
        int total=0;
        for(int p:c.pesos) total+=p;

        int r=GetRandomValue(0,total-1);
        for(int i=0;i<NUM_TIPOS;i++){
            if(r<c.pesos[i]) return i;
            r-=c.pesos[i];
        }
        return 0;
    }
}

void spawnObstaculos(ListaObstaculos& out,float velocidad,const ConfigDificultad& cfg){
    int lado=GetRandomValue(0,3); //0=izq, 1=der, 2=arriba, 3=abajo
    float vel=velocidad*aleatorio(0.8F,1.3F);
    switch(elegirTipo(cfg)){
        case T_PARED_SIMPLE: paredSimple(out,lado,vel); break;
        case T_PARED_DOBLE: paredDoble(out,lado,vel); break;
        case T_PARED_DIAGONAL: paredDiagonal(out,lado,vel); break;
        default: circuloOnda(out,lado,vel); break;
    }
}

void SpawnQueue::add(ListaObstaculos& nuevos,float retraso){
    for(auto& o:nuevos) lista.push_back(Pendiente{move(o),retraso,retraso});
    nuevos.clear();
}

void SpawnQueue::update(float dt,ListaObstaculos& activos){
    for(auto it=lista.begin();it!=lista.end();){
        it->restante-=dt;
        if(it->restante<=0){
            activos.push_back(move(it->obs));
            it=lista.erase(it);
        }else it++;
    }
}

void SpawnQueue::draw() const {
    const Rectangle a=Arena::area();
    const float ax2=a.x+a.width;
    const float ay2=a.y+a.height;
    const float prof=30.0F; //grosor de la banda
    const float tol=1.0F;
    const float asomo=120.0F; //cuanto de la silueta se muestra dentro del area

    for(const auto& p:lista){
        Rectangle b=p.obs->getBounds();
        float urgencia=1.0F-p.restante/p.total; //0 al empezar, 1 al salir
        float pulso=0.5F+0.5F*sin((float)GetTime()*(8.0F+14.0F*urgencia));
        Color col=Fade(ORANGE,0.55F+0.4F*pulso);
        Color borde=Fade(RED,0.9F);

        //tramo del borde que ocupara (recortado al area)
        float x0=max(b.x,a.x),x1=min(b.x+b.width,ax2);
        float y0=max(b.y,a.y),y1=min(b.y+b.height,ay2);

        Rectangle r;
        Vector2 flecha,desp;
        float rot;
        if(b.x+b.width<=a.x+tol){ //entra por la izquierda
            r={a.x,y0,prof,y1-y0};
            flecha={a.x+prof+22.0F,(y0+y1)/2};
            rot=0.0F;
            desp={(a.x-(b.x+b.width))+min(b.width,asomo),0.0F};
        }else if(b.x>=ax2-tol){ //derecha
            r={ax2-prof,y0,prof,y1-y0};
            flecha={ax2-prof-22.0F,(y0+y1)/2};
            rot=180.0F;
            desp={-((b.x-ax2)+min(b.width,asomo)),0.0F};
        }else if(b.y+b.height<=a.y+tol){ //arriba
            r={x0,a.y,x1-x0,prof};
            flecha={(x0+x1)/2,a.y+prof+22.0F};
            rot=90.0F;
            desp={0.0F,(a.y-(b.y+b.height))+min(b.height,asomo)};
        }else{ //abajo
            r={x0,ay2-prof,x1-x0,prof};
            flecha={(x0+x1)/2,ay2-prof-22.0F};
            rot=270.0F;
            desp={0.0F,-((b.y-ay2)+min(b.height,asomo))};
        }

        p.obs->drawGhost(desp,Fade(ORANGE,0.25F+0.2F*pulso),borde); //forma real
    }
}

void CircleObstacle::update(float dt){
    tiempo+=dt;
    base.x+=vel.x*dt;
    base.y+=vel.y*dt;

    float desvio=amplitud*sin(frecuencia*tiempo);
    pos.x=base.x+perp.x*desvio;
    pos.y=base.y+perp.y*desvio;
}

void CircleObstacle::draw() const {
    DrawCircleV(pos,radio,DARKGRAY);
}

bool CircleObstacle::collidesWith(Vector2 centro,float r) const {
    return CheckCollisionCircles(pos,radio,centro,r);
}

    Rectangle a=Arena::area();
    if(vel.x>0 && pos.x-radio>a.x+a.width) return true;

    if(vel.y>0 && pos.y-radio>a.y+a.height) return true;
    if(vel.y<0 && pos.y+radio<a.y) return true;

bool DiagonalWall::collidesWith(Vector2 cj,float radio) const {
    float rad=-angulo*DEG2RAD;
    float dx=cj.x-centro.x;
    float dy=cj.y-centro.y;

    Vector2 local={dx*cos(rad) - dy*sin(rad),dx*sin(rad) + dy*cos(rad)};

    Rectangle rectLocal={-tam.x/2,-tam.y/2,tam.x,tam.y};
    return CheckCollisionCircleRec(local,radio,rectLocal);
}

    if(vel.y>0 && centro.y-R>a.y+a.height) return true;

void DiagonalWall::drawGhost(Vector2 d,Color relleno,Color borde) const {
    Vector2 c={centro.x+d.x,centro.y+d.y};
    DrawRectanglePro(Rectangle{c.x,c.y,tam.x,tam.y},Vector2{tam.x/2,tam.y/2},angulo,relleno);

    float rad=angulo*DEG2RAD;
    float cs=cos(rad);
    float sn=sin(rad);
    const float sx[4]={-1,1,1,-1};
    const float sy[4]={-1,-1,1,1};
    Vector2 p[4];
    for(int i=0;i<4;i++){
        float lx=sx[i]*tam.x/2;
        float ly=sy[i]*tam.y/2;
        p[i]={c.x+lx*cs-ly*sn,c.y+lx*sn+ly*cs};

}

    float rad=angulo*DEG2RAD;
    float c=fabs(cos(rad));
    float s=fabs(sin(rad));
    float hx=tam.x/2*c+tam.y/2*s;
    float hy=tam.x/2*s+tam.y/2*c;

    DrawRectangle((int)pos.x-8,(int)pos.y-2,16,4,WHITE);
}

}

    pickups.clear();
}

    desafio=0.0F;
}

void PowerUpManager::update(float dt,bool activo,Context& ctx){
    //aparicion
    if(activo && ctx.cfg.vidaCada>0){
        proxVida-=dt;
        if(proxVida<=0){
            bool hayVida=false;
            for(const auto& p:pickups) if(p.getTipo()==P_VIDA) hayVida=true;
            if(!hayVida && desafio<=0 && ctx.jugador.getVidas()<ctx.jugador.getMaxVidas())
                pickups.push_back(Pickup(P_VIDA,Arena::puntoLibre(ctx.jugador.getPos(),200.0F,40.0F)));
            proxVida=variar(ctx.cfg.vidaCada);
        }
    }

    //recoger o caducar
    for(auto it=pickups.begin();it!=pickups.end();){
        it->update(dt);
        if(it->collidesWith(ctx.jugador.getPos(),ctx.jugador.getRadio())){
            switch(it->getTipo()){
                case P_VIDA: desafio=TIEMPO_DESAFIO; break;
            }
            it=pickups.erase(it);
        }else if(it->haExpirado()) it=pickups.erase(it);
        else it++;
    }

    //vida
    if(desafio>0){
        desafio-=dt;
        if(desafio<=0){
            desafio=0.0F;
            if(!ctx.jugador.ganarVida()) ctx.puntos+=puntosPor(200,ctx.cfg,ctx.nivel); //ya tenia el maximo
        }
    }
}

void PowerUpManager::dibujar(const Player& jugador) const {
    for(const auto& p:pickups) p.draw();
    if(desafio>0) DrawRing(jugador.getPos(),jugador.getRadio()+5,jugador.getRadio()+8,-90.0F,-90.0F+360.0F*(desafio/TIEMPO_DESAFIO),32,GREEN);
}

void dibujarHud(int puntos,const char* dificultad,int nivel,float tiempo){
    const int W=GetScreenWidth();
    const int alto=(int)Arena::HUD_ALTO;
    const int tam=28;
    const int y=(alto-tam)/2;
    const int margen=20;

    DrawRectangle(0,0,W,alto,BLACK);

    //Puntaje
    DrawText(TextFormat("Puntaje: %d",puntos),margen,y,tam,WHITE);

    //Tiempo
    int min=(int)tiempo/60;
    int seg=(int)tiempo%60;
    const char* textTiempo=TextFormat("%02d:%02d",min,seg);
    DrawText(textTiempo,W/4-MeasureText(textTiempo,tam)/2,y,tam,WHITE);

    //Dificultad
    DrawText(dificultad,(W-MeasureText(dificultad,tam))/2,y,tam,WHITE);

    //Nivel
    const char* textNivel=TextFormat("Nivel %d",nivel);
    DrawText(textNivel,W-MeasureText(textNivel,tam)-margen,y,tam,WHITE);
}

void dibujarCargas(const Player& jugador){
    const int W=GetScreenWidth();
    const int r=14;
    const int sep=44;
    int n=jugador.getMaxCargas();
    int x0=W*3/4-(n-1)*sep/2;
    float y=Arena::HUD_ALTO/2;
    for(int i=0;i<n;i++){
        Vector2 c={(float)(x0+i*sep),y};
        if(i<jugador.getCargas()) DrawCircleV(c,r,SKYBLUE);
        else{
            DrawCircleLinesV(c,r,SKYBLUE);
            if(i==jugador.getCargas()) DrawCircleSector(c,r,-90.0F,-90.0F+360.0F*jugador.getProgresoRecarga(),32,Fade(SKYBLUE,0.5F));
        }
    }
}

    DrawRectangleGradientH((int)a.x,(int)a.y,g,(int)a.height,r,t); //izquierda
    DrawRectangleGradientH((int)(a.x+a.width)-g,(int)a.y,g,(int)a.height,t,r); //derecha
}
